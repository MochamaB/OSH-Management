```csharp
// EmployeeController.cs - Index Action
public async Task<IActionResult> Index(string? search, int? departmentId, int page = 1)
{
    var query = _context.Employees
        .Where(e => e.EmploymentStatus == "Active")
        .AsQueryable();

    // Apply scope
    query = _scopeFilterService.ApplyScope(query, CurrentScope);

    // Includes
    query = query
        .Include(e => e.Station)
        .Include(e => e.Department);

    // Filters
    if (!string.IsNullOrWhiteSpace(search))
        query = query.Where(e =>
            e.FirstName.Contains(search) ||
            e.LastName.Contains(search));

    // Pagination
    var pageSize = 15;
    var employees = await query
        .OrderBy(e => e.LastName)
        .ToPagedListAsync(page, pageSize);

    return View(employees);
}
```

```csharp
// Controllers/OshPolicyController.cs
[Authorize]
public class OshPolicyController : ScopedController
{
    public OshPolicyController(
        OshDbContext context,
        IScopeFilterService scopeFilter,
        ILogger<OshPolicyController> logger)
        : base(context, scopeFilter, logger)
    {
    }

    // GET: OshPolicy/Index
    public async Task<IActionResult> Index(string? search, string? status, int page = 1)
    {
        var query = _context.OshPolicies.AsQueryable();

        // CRITICAL: Apply scope FIRST
        query = _scopeFilterService.ApplyScope(query, CurrentScope);

        // Includes
        query = query.Include(p => p.Station);

        // Filters
        if (!string.IsNullOrWhiteSpace(status))
            query = query.Where(p => p.PolicyStatus == status);

        var pageSize = 15;
        var policies = await query
            .OrderByDescending(p => p.CreatedAt)
            .ToPagedListAsync(page, pageSize);

        return View(policies);
    }

    // GET: OshPolicy/Create
    public async Task<IActionResult> Create()
    {
        // Get stations for dropdown
        var stationsQuery = _context.Stations.AsQueryable();
        stationsQuery = _scopeFilterService.ApplyScope(stationsQuery, CurrentScope);

        ViewBag.Stations = await stationsQuery
            .OrderBy(s => s.StationName)
            .ToListAsync();

        return View();
    }

    // POST: OshPolicy/Create
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Create(OshPolicy policy)
    {
        if (!ModelState.IsValid)
            return View(policy);

        // Verify user can access station
        if (!await CanAccessStationAsync(policy.StationId))
            return ForbidWithLog("Access denied to station");

        policy.CreatedAt = DateTime.UtcNow;
        policy.UpdatedAt = DateTime.UtcNow;

        _context.OshPolicies.Add(policy);
        await _context.SaveChangesAsync();

        // Update OrgMetadata
        await UpdateOrgMetadataAsync(policy.StationId);

        TempData["SuccessMessage"] = "Policy created successfully.";
        return RedirectToAction(nameof(Index));
    }

    private async Task<bool> CanAccessStationAsync(int stationId)
    {
        var station = await _context.Stations.FindAsync(stationId);
        if (station == null) return false;

        // Check if station is in user's scope
        var query = _context.Stations.Where(s => s.StationId == stationId);
        query = _scopeFilterService.ApplyScope(query, CurrentScope);

        return await query.AnyAsync();
    }

    private async Task UpdateOrgMetadataAsync(int stationId)
    {
        var metadata = await _context.OrgMetadata
            .FirstOrDefaultAsync(m => m.StationId == stationId);

        if (metadata == null)
        {
            metadata = new OrgMetadata { StationId = stationId };
            _context.OrgMetadata.Add(metadata);
        }

        metadata.OshPolicyExists = true;
        metadata.ComplianceStatus = "UnderReview";
        metadata.UpdatedAt = DateTime.UtcNow;

        await _context.SaveChangesAsync();
    }
}
```

## 14. KEY TAKEAWAYS

1. **MVC Pattern**: Use Controllers + Views (NOT Razor Pages)
2. **Scope Filtering**: ALWAYS apply via `_scopeFilterService` before includes
3. **Inherit ScopedController**: All feature controllers must inherit
4. **Direct DbContext**: No repository pattern, access `_context` directly
5. **Sidebar Menu**: Hardcoded in `_Sidebar.cshtml` (update manually)
6. **Validation**: FluentValidation auto-registered
7. **Media**: Use polymorphic MediaAssociation pattern
8. **Services**: Register in Program.cs, inject into controllers
